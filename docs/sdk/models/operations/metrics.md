# Metrics

Metrics aggregated across a workflow for a given time window.

## Example Usage

```typescript
import { Metrics } from "circleci-v2-sdk/sdk/models/operations";

let value: Metrics = {
  completedRuns: 202796,
  durationMetrics: {
    max: 423588,
    mean: 357758,
    median: 163684,
    min: 441374,
    p95: 262800,
    standardDeviation: 5220.62,
  },
  failedRuns: 35160,
  mttr: 906232,
  successRate: 8163.64,
  successfulRuns: 552581,
  throughput: 8518.09,
  totalCreditsUsed: 962396,
  totalRuns: 110522,
  windowEnd: new Date("2023-11-23T09:23:59.598Z"),
  windowStart: new Date("2024-12-29T19:14:03.123Z"),
};
```
