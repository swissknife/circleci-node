# Metrics

Metrics aggregated across a workflow for a given time window.

## Example Usage

```typescript
import { Metrics } from "circleci-v2-sdk/sdk/models/operations";

let value: Metrics = {
  completedRuns: 411615,
  durationMetrics: {
    max: 489459,
    mean: 243678,
    median: 588639,
    min: 753240,
    p95: 235834,
    standardDeviation: 6205.00,
  },
  failedRuns: 639622,
  mttr: 948541,
  successRate: 7782.76,
  successfulRuns: 848346,
  throughput: 4904.20,
  totalCreditsUsed: 185348,
  totalRuns: 995816,
  windowEnd: new Date("2022-05-22T18:34:26.178Z"),
  windowStart: new Date("2023-06-01T17:12:47.982Z"),
};
```
